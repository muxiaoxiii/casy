## 三、期限计算引擎修正

### 3.1 add_workdays 函数

实现专利法实施细则第5条的三项规则：
1. **起算日不计**：期限从起算日的次日开始计算
2. **休假日顺延**：期限届满日为法定休假日的，顺延到休假日后的第一个工作日
3. **月末处理**：按月计算期限时，若起算日对应日不存在，以月末为届满日

```rust
// src-tauri/src/formula/holidays.rs — HolidayCalendar impl

use chrono::{NaiveDate, Duration, Datelike};

pub struct HolidayCalendar {
    pub holidays: HashSet<NaiveDate>,
    /// 调休上班日（周末但实际为工作日）
    pub workdays: HashSet<NaiveDate>,
}

impl HolidayCalendar {
    /// 判断某日是否为工作日（排除周末和法定假日，含调休上班日）
    pub fn is_workday(&self, date: NaiveDate) -> bool {
        // 调休上班日优先视为工作日
        if self.workdays.contains(&date) {
            return true;
        }
        // 周末视为非工作日
        let weekday = date.weekday();
        if weekday == Weekday::Sat || weekday == Weekday::Sun {
            return false;
        }
        // 法定假日视为非工作日
        !self.holidays.contains(&date)
    }

    /// 从 date 之后找到下一个工作日（不含 date 本身）
    fn next_workday_after(&self, date: NaiveDate) -> NaiveDate {
        let mut d = date + Duration::days(1);
        while !self.is_workday(d) {
            d = d + Duration::days(1);
        }
        d
    }

    /// 如果 date 是非工作日，顺延到之后的第一个工作日；否则原样返回
    fn extend_to_workday(&self, date: NaiveDate) -> NaiveDate {
        if self.is_workday(date) {
            date
        } else {
            self.next_workday_after(date - Duration::days(1))
        }
    }

    /// 计算工作日届满日
    ///
    /// 规则（专利法实施细则第5条）：
    /// 1. 起算日不计入，从次日起算
    /// 2. 顺数 N 个工作日
    /// 3. 若届满日落在休假日，顺延到休假日后第一个工作日
    pub fn add_workdays(&self, start: NaiveDate, n: i64) -> NaiveDate {
        // 起算日不计入，从次日开始
        let mut current = start + Duration::days(1);
        let mut remaining = n;

        while remaining > 0 {
            // 跳过非工作日
            while !self.is_workday(current) {
                current = current + Duration::days(1);
            }
            // 消耗一个工作日
            remaining -= 1;
            if remaining > 0 {
                current = current + Duration::days(1);
            }
        }

        // 确保最终日期是工作日（休假日顺延）
        self.extend_to_workday(current)
    }

    /// 按月计算届满日
    ///
    /// 规则（专利法实施细则第5条）：
    /// 1. 起算日不计入，从次日起算
    /// 2. 以次日为基准加 N 个月
    /// 3. 若次日是月末最后一天且目标月无对应日，以目标月月末为届满日
    /// 4. 若届满日落在休假日，顺延到休假日后第一个工作日
    pub fn add_months(&self, start: NaiveDate, months: u32) -> NaiveDate {
        // 起算日不计入，从次日起算
        let from = start + Duration::days(1);
        let due = add_months_clamp(from, months);
        // 休假日顺延
        self.extend_to_workday(due)
    }
}

/// 月份加法，含月末钳制（clamping）
///
/// 例：
/// - 1月30日 + 1个月 = 2月28日（非闰年）/ 2月29日（闰年）
/// - 1月31日 + 1个月 = 2月28日（非闰年）/ 2月29日（闰年）
/// - 3月31日 + 1个月 = 4月30日
fn add_months_clamp(date: NaiveDate, months: u32) -> NaiveDate {
    let total = date.month() + months;
    let year = date.year() + ((total - 1) / 12) as i32;
    let month = ((total - 1) % 12) + 1;
    let max_day = days_in_month(year, month);
    // 钳制到目标月的最大天数（月末处理）
    let day = date.day().min(max_day);
    NaiveDate::from_ymd_opt(year, month, day).unwrap_or(date)
}

fn days_in_month(year: i32, month: u32) -> u32 {
    if month == 12 {
        31
    } else {
        let next = NaiveDate::from_ymd_opt(year, month + 1, 1).unwrap();
        let curr = NaiveDate::from_ymd_opt(year, month, 1).unwrap();
        (next - curr).num_days() as u32
    }
}
```

### 3.2 DeadlineEngine::evaluate_case 修正

期限引擎需统一使用修正后的计算方法：

```rust
// src-tauri/src/formula/engine.rs

impl DeadlineEngine {
    /// 计算单个案件的所有期限
    pub fn evaluate_case(&self, case: &Case, today: NaiveDate) -> Vec<DeadlineResult> {
        self.rules.iter()
            .filter(|r| r.track == case.track || r.track == "all")
            .filter_map(|rule| {
                // 检查额外条件（含 verdict_type、procedure_type 等）
                if let Some(cond) = &rule.condition {
                    let cond: serde_json::Value = serde_json::from_str(cond).ok()?;
                    if !check_condition(case, &cond) { return None; }
                }

                // 获取触发日期
                let trigger = get_case_date_field(case, &rule.trigger_field)?;

                // 计算到期日
                let due = match rule.offset_unit.as_str() {
                    "day" => {
                        // 日历日：起算日不计入 + 休假日顺延
                        let raw = trigger + Duration::days(rule.offset_days);
                        self.calendar.extend_to_workday(raw)
                    },
                    "workday" => {
                        // 工作日：起算日不计入 + 顺数工作日 + 休假日顺延
                        self.calendar.add_workdays(trigger, rule.offset_days)
                    },
                    "month" => {
                        // 按月：起算日不计入 + 月末钳制 + 休假日顺延
                        self.calendar.add_months(trigger, rule.offset_days as u32)
                    },
                    _ => return None,
                };

                let days_left = (due - today).num_days();
                Some(DeadlineResult {
                    rule_name: rule.rule_name.clone(),
                    due_date: due,
                    days_left,
                    urgency: if days_left <= 3 { Urgency::Red }
                             else if days_left <= 14 { Urgency::Yellow }
                             else { Urgency::Green },
                    case_id: case.id.clone(),
                    case_name: case.case_name.clone(),
                })
            })
            .collect()
    }
}

fn check_condition(case: &Case, cond: &serde_json::Value) -> bool {
    // 案由条件
    if let Some(cause) = cond.get("cause_action").and_then(|v| v.as_str()) {
        if case.cause_action.as_deref() != Some(cause) { return false; }
    }
    // 审理程序条件（普通/简易）
    if let Some(procedure) = cond.get("procedure_type").and_then(|v| v.as_str()) {
        if case.procedure_type.as_deref() != Some(procedure) { return false; }
    }
    // 判决/裁定类型条件
    if let Some(vtype) = cond.get("verdict_type").and_then(|v| v.as_str()) {
        if case.verdict_type.as_deref() != Some(vtype) { return false; }
    }
    // 必填字段条件
    if let Some(required_field) = cond.get("required_field").and_then(|v| v.as_str()) {
        let val = match required_field {
            "petitioner_submit_date" => case.petitioner_submit_date.as_ref(),
            "petitioner_received_date" => case.petitioner_received_date.as_ref(),
            "patentee_received_date" => case.patentee_received_date.as_ref(),
            "patentee_received_supp_date" => case.patentee_received_supp_date.as_ref(),
            _ => None,
        };
        if val.is_none() { return false; }
    }
    true
}
```
